const STAT_CATEGORIES = [
  { key: 'career_points', label: 'career points' },
  { key: 'career_rebounds', label: 'career rebounds' },
  { key: 'career_assists', label: 'career assists' },
  { key: 'career_steals', label: 'career steals' },
  { key: 'career_blocks', label: 'career blocks' },
  { key: 'championships', label: 'championships' },
  { key: 'career_ppg', label: 'points per game' },
];

const CATEGORY_PROMPTS = {
  career_points: 'Who scored more career points?',
  career_rebounds: 'Who grabbed more career rebounds?',
  career_assists: 'Who dished more career assists?',
  career_steals: 'Who racked up more career steals?',
  career_blocks: 'Who swatted more career blocks?',
  championships: 'Who won more championships?',
  career_ppg: 'Who averaged more points per game?',
};

function difficultyBand(streak) {
  if (streak < 3) return { min: 0.5, max: 0.9 };
  if (streak < 6) return { min: 0.28, max: 0.5 };
  if (streak < 10) return { min: 0.12, max: 0.28 };
  return { min: 0.03, max: 0.12 };
}

function pairKey(idA, idB) {
  return idA < idB ? `${idA}-${idB}` : `${idB}-${idA}`;
}

function generateMatchup(playerPool, streak, seenPairs) {
  const categoriesToTry = [...STAT_CATEGORIES].sort(() => Math.random() - 0.5);
  const { min, max } = difficultyBand(streak);

  for (const category of categoriesToTry) {
    const eligible = playerPool.filter((p) => p[category.key] != null);
    if (eligible.length < 6) continue;

    const sorted = [...eligible].sort((a, b) => a[category.key] - b[category.key]);
    const n = sorted.length;
    const minGap = Math.max(1, Math.floor(n * min));
    const maxGap = Math.max(minGap + 1, Math.floor(n * max));

    // Collect every valid pair in the difficulty band, then pick uniformly at
    // random — avoids always landing on the same extreme record-holders.
    const candidates = [];
    for (let i = 0; i < n; i++) {
      for (let gap = minGap; gap <= maxGap; gap++) {
        const j = i + gap;
        if (j >= n) break;
        const a = sorted[i];
        const b = sorted[j];
        if (a[category.key] === b[category.key]) continue;
        if (seenPairs.has(pairKey(a.id, b.id))) continue;
        candidates.push([a, b]);
      }
    }

    if (candidates.length === 0) continue;

    const [a, b] = candidates[Math.floor(Math.random() * candidates.length)];
    const correctPlayerId = a[category.key] > b[category.key] ? a.id : b.id;
    const swap = Math.random() < 0.5;

    return {
      category: category.key,
      categoryLabel: category.label,
      playerA: swap ? a : b,
      playerB: swap ? b : a,
      correctPlayerId,
      pairKey: pairKey(a.id, b.id),
    };
  }

  if (streak !== 0) return generateMatchup(playerPool, 0, seenPairs);
  if (seenPairs.size > 0) return generateMatchup(playerPool, 0, new Set());
  return null;
}

function formatStat(category, value) {
  if (category === 'career_ppg') return `${value} PPG`;
  if (category === 'championships') return `${value} ${value === 1 ? 'ring' : 'rings'}`;
  return value.toLocaleString();
}

// Placeholder avatar — swap for the real Homegrown Hoops avatar component in production.
function PlayerAvatar({ label, size = 96 }) {
  const initials = label.split(' ').map((w) => w[0]).join('').slice(0, 2);
  return (
    <div
      style={{
        width: size, height: size, borderRadius: '50%',
        background: 'linear-gradient(135deg,#1e293b,#334155)',
        display: 'flex', alignItems: 'center', justifyContent: 'center',
        color: '#f97316', fontSize: size * 0.32, fontWeight: 800,
        border: '3px solid #f97316',
      }}
    >
      {initials}
    </div>
  );
}

export default function HigherLowerDemo() {
  const [streak, setStreak] = useState(0);
  const [bestStreak, setBestStreak] = useState(0);
  const [matchup, setMatchup] = useState(null);
  const [seenPairs, setSeenPairs] = useState(new Set());
  const [revealed, setRevealed] = useState(null);
  const [gameOver, setGameOver] = useState(false);

  const nextRound = useCallback((currentStreak, pairs) => {
    const next = generateMatchup(PLAYERS, currentStreak, pairs);
    setMatchup(next);
    setRevealed(null);
  }, []);

  useEffect(() => {
    nextRound(0, new Set());
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  function handlePick(pickedId) {
    if (revealed) return;
    const correct = pickedId === matchup.correctPlayerId;
    setRevealed(correct ? 'correct' : 'wrong');

    if (correct) {
      const newStreak = streak + 1;
      setStreak(newStreak);
      setBestStreak((b) => Math.max(b, newStreak));
      const updated = new Set(seenPairs);
      updated.add(matchup.pairKey);
      setSeenPairs(updated);
      setTimeout(() => nextRound(newStreak, updated), 900);
    } else {
      setGameOver(true);
    }
  }

  function handlePlayAgain() {
    setStreak(0);
    setGameOver(false);
    setSeenPairs(new Set());
    nextRound(0, new Set());
  }

  if (!matchup) return null;

  const { playerA, playerB, category, correctPlayerId } = matchup;

  return (
    <div style={{ minHeight: '100vh', background: '#0f172a', display: 'flex', alignItems: 'center', justifyContent: 'center', fontFamily: 'system-ui, sans-serif', padding: 16 }}>
      <div style={{ width: '100%', maxWidth: 440, background: '#1e293b', borderRadius: 20, padding: 24, color: 'white', boxShadow: '0 20px 40px rgba(0,0,0,0.4)' }}>
        <div style={{ textAlign: 'center', fontSize: 12, letterSpacing: 1, color: '#94a3b8', marginBottom: 4 }}>HIGHER OR LOWER</div>

        {gameOver ? (
          <div style={{ textAlign: 'center', padding: '24px 0' }}>
            <div style={{ fontSize: 40, marginBottom: 8 }}>🏀</div>
            <h2 style={{ margin: '0 0 8px' }}>Run over!</h2>
            <p style={{ color: '#cbd5e1', margin: '4px 0' }}>You reached a streak of <b>{streak}</b></p>
            <p style={{ color: '#cbd5e1', margin: '4px 0' }}>Best streak: <b>{bestStreak}</b></p>
            <button
              onClick={handlePlayAgain}
              style={{ marginTop: 20, padding: '12px 28px', borderRadius: 10, border: 'none', background: '#f97316', color: 'white', fontWeight: 700, fontSize: 15, cursor: 'pointer' }}
            >
              Play again
            </button>
          </div>
        ) : (
          <>
            <div style={{ display: 'flex', justifyContent: 'space-between', fontWeight: 700, marginBottom: 16, fontSize: 14 }}>
              <span>Streak: {streak}</span>
              <span style={{ color: '#f97316' }}>Best: {bestStreak}</span>
            </div>

            <h3 style={{ textAlign: 'center', margin: '0 0 20px', fontSize: 18 }}>{CATEGORY_PROMPTS[category]}</h3>

            <div style={{ display: 'flex', gap: 12 }}>
              {[playerA, playerB].map((p) => {
                const isCorrect = p.id === correctPlayerId;
                const state = revealed ? (isCorrect ? 'correct' : 'incorrect') : null;
                return (
                  <button
                    key={p.id}
                    onClick={() => handlePick(p.id)}
                    disabled={!!revealed}
                    style={{
                      flex: 1, display: 'flex', flexDirection: 'column', alignItems: 'center', gap: 10,
                      padding: 16, borderRadius: 14, cursor: revealed ? 'default' : 'pointer',
                      border: state === 'correct' ? '2px solid #22c55e' : state === 'incorrect' ? '2px solid #ef4444' : '2px solid #334155',
                      background: state === 'correct' ? 'rgba(34,197,94,0.12)' : state === 'incorrect' ? 'rgba(239,68,68,0.12)' : '#0f172a',
                      transition: 'all 0.2s',
                    }}
                  >
                    <PlayerAvatar label={p.full_name} />
                    <div style={{ fontWeight: 700, fontSize: 13, textAlign: 'center' }}>{p.full_name}</div>
                    <div style={{ fontSize: 11, color: '#64748b' }}>{p.era_label}</div>
                    {revealed && (
                      <div style={{ fontSize: 15, fontWeight: 800, color: state === 'correct' ? '#22c55e' : '#ef4444' }}>
                        {formatStat(category, p[category])}
                      </div>
                    )}
                  </button>
                );
              })}
            </div>
          </>
        )}
      </div>
    </div>
  );
}
